== ut_hold.py
"""Hold ut-browser's engine build lock through ut-browser's own code, as its build_wasm.py does."""
import sys
import time
from datetime import datetime

sys.dont_write_bytecode = True
sys.path.insert(0, "G:/Claude Stuff/ut-browser/scripts")
import buildlock  # noqa: E402


def log(msg):
    print(f"{datetime.now().isoformat(timespec='seconds')} ut-browser: {msg}", flush=True)


hold = float(sys.argv[1])
log(f"asking for the engine build lock in {buildlock.cache_dir()}")
with buildlock.machine_lock("ut-browser: the engine cone (keeperfxbrowser lock proof)"):
    log(f"holding the engine build lock for {hold:.0f}s, as a compile would")
    time.sleep(hold)
    log("done, releasing the engine build lock")
log("released")
